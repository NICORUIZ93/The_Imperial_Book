- Mover puntos de suma y de bifurcación sin alterar el sistema.

```python
# verificacion.py
from sympy import symbols, simplify

G1, G2, H1, H2 = symbols("G1 G2 H1 H2")
P1 = G1 * G2
L1, L2 = -G1 * H1, -G2 * H2
delta = 1 - (L1 + L2) + L1 * L2
T = simplify(P1 / delta)
print("T:", T)
print("coincide_con_factorizado:", simplify((1 + G1 * H1) * (1 + G2 * H2) - delta) == 0)
```
